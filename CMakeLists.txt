cmake_minimum_required(VERSION 3.20)
project(LocalHistoria VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build configuration" FORCE)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SQLite3 REQUIRED)
find_package(Drogon CONFIG REQUIRED)
find_package(jsoncpp CONFIG REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_library(historia_core src/core/demands.cpp src/core/domestic.cpp src/core/lifecycle.cpp src/core/simulation.cpp src/core/politics.cpp src/store/store.cpp src/store/event_log.cpp
  src/world/delta.cpp src/world/delta_checks.cpp src/world/geometry.cpp src/world/schema.cpp
  src/world/compiler/compiler.cpp src/world/compiler/helpers.cpp src/world/compiler/identity_ops.cpp
  src/world/compiler/treaty_ops.cpp src/world/compiler/treaty_terms.cpp src/world/compiler/proposals.cpp
  src/world/compiler/force_ops.cpp src/world/compiler/territory_ops.cpp
  src/world/compiler/events.cpp src/world/compiler/domestic.cpp
  src/status/helpers.cpp src/status/conflicts.cpp src/status/control.cpp src/status/force.cpp src/status/seed.cpp src/status/national.cpp
  src/status/agreements.cpp src/status/regimes.cpp src/status/monthly/monthly_status.cpp
  src/status/monthly/fronts.cpp src/status/monthly/casualties.cpp src/status/monthly/economy.cpp src/status/monthly/recovery.cpp
  src/status/monthly/unrest.cpp src/status/monthly/rebellion.cpp)
target_include_directories(historia_core PUBLIC include PRIVATE src)
if(TARGET SQLite3::SQLite3)
  set(SQLITE_TARGET SQLite3::SQLite3)
else()
  set(SQLITE_TARGET SQLite::SQLite3)
endif()
target_link_libraries(historia_core PUBLIC JsonCpp::JsonCpp ${SQLITE_TARGET})
if(NOT MSVC)
  target_compile_options(historia_core PRIVATE -Wall -Wextra -Wpedantic -Wshadow)
endif()
add_library(historia_services src/inference/transport.cpp src/inference/prompts.cpp src/inference/history.cpp src/turn/adjudication_input.cpp src/turn/plan.cpp
  src/turn/run.cpp
  src/inference/references.cpp src/server/campaign_factory.cpp src/server/routes.cpp
  src/server/campaign_handlers.cpp src/server/world_handlers.cpp)
target_include_directories(historia_services PRIVATE src)
target_link_libraries(historia_services PUBLIC historia_core Drogon::Drogon)
if(NOT MSVC)
  target_compile_options(historia_services PRIVATE -Wall -Wextra -Wpedantic -Wshadow)
endif()
add_executable(localhistoria src/main.cpp)
target_include_directories(localhistoria PRIVATE src)
target_link_libraries(localhistoria PRIVATE historia_services)
enable_testing()
add_executable(core_tests tests/core/main.cpp tests/core/history_tests.cpp)
target_link_libraries(core_tests PRIVATE historia_services)
target_compile_definitions(core_tests PRIVATE FIXTURE_DIR="${CMAKE_SOURCE_DIR}/tests/fixtures")
add_test(NAME core COMMAND core_tests)
add_test(NAME data COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/data_test.py)
add_executable(replay_verify tools/replay_verify.cpp)
target_link_libraries(replay_verify PRIVATE historia_core)
add_executable(world_tests tests/world/main.cpp tests/world/deltas.cpp tests/world/operations.cpp
  tests/world/movement.cpp tests/world/naming.cpp tests/world/treaties.cpp tests/world/events.cpp)
target_link_libraries(world_tests PRIVATE historia_services)
add_test(NAME world COMMAND world_tests)
add_executable(context_tests tests/context_tests.cpp)
target_link_libraries(context_tests PRIVATE historia_services)
add_test(NAME context COMMAND context_tests)
add_executable(world_probe tools/world_probe.cpp)
target_link_libraries(world_probe PRIVATE historia_services)
add_executable(politics_tests tests/politics_tests.cpp)
target_link_libraries(politics_tests PRIVATE historia_services)
add_test(NAME politics COMMAND politics_tests)
add_executable(status_tests tests/status/main.cpp tests/status/real_world.cpp tests/status/unrest.cpp tests/status/conquest.cpp tests/status/forces.cpp tests/status/displacement.cpp tests/status/economy.cpp tests/status/pacing.cpp)
target_link_libraries(status_tests PRIVATE historia_core)
target_compile_definitions(status_tests PRIVATE SOURCE_DIR="${CMAKE_SOURCE_DIR}")
add_test(NAME status COMMAND status_tests)
add_executable(military_tests tests/military_tests.cpp)
target_link_libraries(military_tests PRIVATE historia_core)
target_compile_definitions(military_tests PRIVATE SOURCE_DIR="${CMAKE_SOURCE_DIR}")
add_test(NAME military COMMAND military_tests)
add_executable(turn_tests tests/turn_tests.cpp)
target_link_libraries(turn_tests PRIVATE historia_services)
target_include_directories(turn_tests PRIVATE src)
add_test(NAME turn COMMAND turn_tests)

foreach(version 0 2 3 4)
  add_test(NAME legacy_replay_v${version} COMMAND replay_verify
    ${CMAKE_SOURCE_DIR}/tests/fixtures/legacy-lifecycle-v${version}.json)
endforeach()
add_executable(lifecycle_tests tests/lifecycle_tests.cpp)
target_link_libraries(lifecycle_tests PRIVATE historia_core)
target_compile_definitions(lifecycle_tests PRIVATE FIXTURE_DIR="${CMAKE_SOURCE_DIR}/tests/fixtures")
add_test(NAME lifecycle COMMAND lifecycle_tests)

add_test(NAME mixed_replay COMMAND replay_verify ${CMAKE_BINARY_DIR}/mixed-lifecycle-export.json)
set_tests_properties(mixed_replay PROPERTIES FIXTURES_REQUIRED mixed_lifecycle)
add_executable(consent_tests tests/consent_tests.cpp)
target_link_libraries(consent_tests PRIVATE historia_core)
target_compile_definitions(consent_tests PRIVATE FIXTURE_DIR="${CMAKE_SOURCE_DIR}/tests/fixtures")
add_test(NAME consent COMMAND consent_tests)
add_executable(conflict_tests tests/conflict_tests.cpp)
target_link_libraries(conflict_tests PRIVATE historia_core)
target_compile_definitions(conflict_tests PRIVATE FIXTURE_DIR="${CMAKE_SOURCE_DIR}/tests/fixtures")
add_test(NAME conflicts COMMAND conflict_tests)
add_executable(access_tests tests/access_tests.cpp)
target_link_libraries(access_tests PRIVATE historia_core)
add_test(NAME access COMMAND access_tests)
add_executable(domestic_benchmark tools/domestic_benchmark.cpp)
target_link_libraries(domestic_benchmark PRIVATE historia_core)
add_executable(domestic_tests tests/domestic_tests.cpp)
target_link_libraries(domestic_tests PRIVATE historia_core)
add_test(NAME domestic COMMAND domestic_tests)
set_tests_properties(domestic PROPERTIES FIXTURES_SETUP domestic_replay_fixture)
add_test(NAME domestic_replay COMMAND replay_verify ${CMAKE_BINARY_DIR}/domestic-replay.json)
set_tests_properties(domestic_replay PROPERTIES FIXTURES_REQUIRED domestic_replay_fixture)
add_executable(demand_tests tests/demand_tests.cpp)
target_link_libraries(demand_tests PRIVATE historia_core)
add_test(NAME demands COMMAND demand_tests)
set_tests_properties(demands PROPERTIES FIXTURES_SETUP demand_replay_fixture)
add_test(NAME demand_replay COMMAND replay_verify ${CMAKE_BINARY_DIR}/demand-replay.json)
set_tests_properties(demand_replay PROPERTIES FIXTURES_REQUIRED demand_replay_fixture)
# Every suite runs in well under a second; a hang fails instead of running forever.
get_property(all_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${all_tests} PROPERTIES TIMEOUT 60)
